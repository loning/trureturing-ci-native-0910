def elementaryCoeff (p : Real[X]) (k : Nat) : Real :=
  (-1)^k * p.coeff (4-k)
def weight (alpha : Real) (k : Nat) : Real :=
  (descPochhammer Real k).eval 4 * (descPochhammer Real k).eval (4+alpha)
def normalizedCoeff (alpha : Real) (p : Real[X]) (k : Nat) : Real :=
  elementaryCoeff p k / weight alpha k
def convolutionCoeff (alpha : Real) (p q : Real[X]) (k : Nat) : Real :=
  weight alpha k * ((Finset.range (k+1)).sum fun i =>
    normalizedCoeff alpha p i * normalizedCoeff alpha q (k-i))
def boxplus4 (alpha : Real) (p q : Real[X]) : Real[X] :=
  C (convolutionCoeff alpha p q 0) * X^4 -
    C (convolutionCoeff alpha p q 1) * X^3 +
    C (convolutionCoeff alpha p q 2) * X^2 -
    C (convolutionCoeff alpha p q 3) * X + C (convolutionCoeff alpha p q 4)
def rootFour (a b c d : Real) : Real[X] :=
  (X-C a)*(X-C b)*(X-C c)*(X-C d)

theorem conditional_hankel_projection (roots : Fin 4 -> Complex)
    (hc : forall z, z ∈ Finset.univ.image roots ↔ conj z ∈ Finset.univ.image roots)
    (hpsd : Matrix.PosSemidef (newtonHankel roots)) :
    forall j, ∃ r : Real, roots j = (r : Complex) := by
  exact (newtonHankel_posSemidef_iff_roots_real roots hc).mp hpsd
